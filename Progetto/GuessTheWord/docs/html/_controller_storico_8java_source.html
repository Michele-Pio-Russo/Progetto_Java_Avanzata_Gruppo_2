106 private void configuraTabella() {
107 dateCol.setCellValueFactory(c ->
108 new SimpleStringProperty(c.getValue().getDataPartita()));
109 opponentCol.setCellValueFactory(c ->
110 new SimpleStringProperty(c.getValue().getAvversario()));
111 wordCol.setCellValueFactory(c ->
112 new SimpleStringProperty(c.getValue().getParolaCorretta()));
113 outcomeCol.setCellValueFactory(c ->
114 new SimpleStringProperty(formatOutcome(c.getValue().getEsito())));
115 timeCol.setCellValueFactory(c ->
116 new SimpleLongProperty(c.getValue().getTempoRispostaMs()).asObject());
117
118 outcomeCol.setCellFactory(col -> new TableCell<VoceStorico, String>() {
119 @Override
120 protected void updateItem(String item, boolean empty) {
121 super.updateItem(item, empty);
122 if (empty || item == null) {
123 setText(null);
124 setStyle("");
125 } else {
126 setText(item);
127 if (item.startsWith("Vinto")) {
128 setStyle("-fx-text-fill: #3fb950; -fx-font-weight: bold;");
129 } else if (item.startsWith("Perso")) {
130 setStyle("-fx-text-fill: #f85149; -fx-font-weight: bold;");
131 } else {
132 setStyle("-fx-text-fill: #e3b341; -fx-font-weight: bold;");
133 }
134 }
135 }
136 });
137
138 timeCol.setCellFactory(col -> new TableCell<VoceStorico, Long>() {
139 @Override
140 protected void updateItem(Long item, boolean empty) {
141 super.updateItem(item, empty);
142 if (empty || item == null) {
143 setText(null);
144 } else {
145 setText(item < 0 ? "\u2014" : String.valueOf(item));
146 }
147 }

205 private String formatOutcome(String esito) {
206 if (esito == null) return "\u2014";
207 if ("win".equalsIgnoreCase(esito)) return "Vinto";
208 if ("loss".equalsIgnoreCase(esito)) return "Perso";
209 if ("draw".equalsIgnoreCase(esito)) return "Pareggio";
210 return esito;
211 }

217 private void tornaAlLogin(String motivo) {

227 Alert alert = new Alert(Alert.AlertType.ERROR);

229 alert.setHeaderText("Connessione interrotta");
