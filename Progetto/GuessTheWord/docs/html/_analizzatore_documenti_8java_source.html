52 public void analyze(List<File> files) throws IOException {
53 // Lettura e concatenazione di tutti i file
54 StringBuilder sb = new StringBuilder();
55 for (File f : files) {
56 byte[] bytes = Files.readAllBytes(f.toPath());
57 sb.append(new String(bytes, "UTF-8")).append(" ");
58 }
59 aggregatedText = sb.toString();
60
61 // Tokenizzazione: lowercase + rimozione punteggiatura + split
62 String[] tokens = aggregatedText
63 .toLowerCase()
64 .replaceAll("[^a-zA-Z\\s]", " ")
65 .split("\\s+");
66
67 // Conteggio totale parole valide con Stream
68 final long totalWords = Arrays.stream(tokens)
69 .filter(t -> !t.isEmpty())
70 .count();
71
72 if (totalWords == 0) {
73 LOGGER.warning("Nessuna parola trovata nei documenti selezionati.");
74 return;
75 }
76
77 // Pipeline Stream:
78 // filter → groupingBy (counting) → entrySet stream → toMap (calcolo TF)
79 termFrequency = Arrays.stream(tokens)
80 .filter(t -> !t.isEmpty() && t.length() > 2)
81 .collect(Collectors.groupingBy(
82 t -> t, // classifier: parola stessa
83 Collectors.counting() // downstream: conta occorrenze
84 ))
85 .entrySet().stream()
86 .collect(Collectors.toMap(
87 Map.Entry::getKey,
88 e -> (double) e.getValue() / totalWords // TF = count / N
89 ));
90
91 LOGGER.info("Analisi completata: " + termFrequency.size()
92 + " parole uniche su " + totalWords + " totali.");
93 }

131 public String extractRandomExcerpt(int numWords) {
132 if (aggregatedText.isEmpty()) return "";
133 String[] allWords = aggregatedText.split("\\s+");
134 if (allWords.length <= numWords) return aggregatedText;
135 Random rnd = new Random();
136 int start = rnd.nextInt(allWords.length - numWords);
137 // Stream per assemblare l'estratto
138 return Arrays.stream(allWords, start, start + numWords)
139 .collect(Collectors.joining(" "));
140 }

153 public String selectWordFromExcerpt(String excerpt, int difficulty) {
154 String[] words = excerpt.toLowerCase()
155 .replaceAll("[^a-z\\s]", "")
156 .split("\\s+");
157
158 // Lambda Comparator: ordina per TF ascendente (difficile) o discendente (facile)
159 final boolean preferRare = (difficulty >= 2);
160 Comparator<String> byTf = (a, b) -> {
161 double tfA = termFrequency.getOrDefault(a, 0.0);
162 double tfB = termFrequency.getOrDefault(b, 0.0);
163 return preferRare
164 ? Double.compare(tfA, tfB) // crescente: parole rare prima
165 : Double.compare(tfB, tfA); // decrescente: parole comuni prima
166 };
167
168 // Pipeline Stream: filter → sorted (lambda Comparator) → limit → collect
169 List<String> candidates = Arrays.stream(words)
170 .filter(w -> w.length() >= 4 && termFrequency.containsKey(w))
171 .sorted(byTf)
172 .limit(5)
173 .collect(Collectors.toList());
174
175 if (candidates.isEmpty()) return null;
176
177 // Selezione casuale tra i top-5 candidati
178 return candidates.get(new Random().nextInt(candidates.size()));
179 }
