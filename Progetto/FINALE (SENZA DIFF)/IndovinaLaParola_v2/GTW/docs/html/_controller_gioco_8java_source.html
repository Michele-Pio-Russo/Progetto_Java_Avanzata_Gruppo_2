151 private void gestisciMessaggioServer(final Messaggio msg) {
152 Platform.runLater(new Runnable() {
153 @Override
154 public void run() {
155 switch (msg.getTipo()) {
156 case WAITING:
157 dettaglioAttesa.setText((String) msg.getCarico());
158 mostraAttesa();
159 break;
160 case CHALLENGE_START:
161 avviaSfida((PayloadSfida) msg.getCarico());
162 break;
163 case CHALLENGE_RESULT:
164 mostraRisultato((RisultatoSfida) msg.getCarico());
165 break;
166 case ERROR:
167 gestisciErroreServer((String) msg.getCarico());
168 break;
169 default:
170 break;
171 }
172 }
173 });
174 }

184 private void avviaSfida(PayloadSfida carico) {
185 tempoInizioSfida = System.currentTimeMillis();
186 secondiRimanenti = carico.getSecondiTimeout();
187
188 areaTesto.setText(carico.getEstrattoTesto());
189 List<String> words = carico.getParoleCifrate();
190 StringBuilder sb = new StringBuilder();
191 for (int i = 0; i < words.size(); i++) {
192 sb.append(words.get(i));
193 if (i < words.size() - 1) sb.append(", ");
194 }
195 etichettaParoleCifrate.setText(sb.toString());
196
197 campoRisposta.clear();
198 campoRisposta.setDisable(false);
199 etichettaFeedback.setText("");
200
201 mostraGioco();
202 etichettaStato.setText("In gioco!");
203
204 avviaCountdown();
205 }

214 private void avviaCountdown() {
215 if (timelineTimer != null) timelineTimer.stop();
216
217 etichettaTimer.setText(String.valueOf(secondiRimanenti));
218 etichettaTimer.setStyle("-fx-text-fill: #f0f6fc; -fx-font-size: 32px; -fx-font-weight: bold;");
219
220 timelineTimer = new Timeline(new KeyFrame(Duration.seconds(1),
221 new EventHandler<ActionEvent>() {
222 @Override
223 public void handle(ActionEvent event) {
224 secondiRimanenti--;
225 etichettaTimer.setText(String.valueOf(Math.max(0, secondiRimanenti)));
226
227 if (secondiRimanenti <= 10) {
228 etichettaTimer.setStyle(
229 "-fx-text-fill: #f85149; -fx-font-size: 32px; -fx-font-weight: bold;");
230 }
231
232 if (secondiRimanenti <= 0) {
233 timelineTimer.stop();
234 campoRisposta.setDisable(true);
235 mostraFeedback("Tempo scaduto!");
236 }
237 }
238 }));
239
240 timelineTimer.setCycleCount(secondiRimanenti);
241 timelineTimer.play();
242 }

252 private void mostraRisultato(RisultatoSfida result) {
253 if (timelineTimer != null) timelineTimer.stop();
254 campoRisposta.setDisable(true);
255 etichettaStato.setText("Partita terminata");
256
257 emojiRisultato.setText(result.getEsito().toEmoji());
258 titoloRisultato.setText(result.getEsito().toDisplayString());
259
260 switch (result.getEsito()) {
261 case VITTORIA:
262 titoloRisultato.setStyle(
263 "-fx-text-fill: #3fb950; -fx-font-size: 26px; -fx-font-weight: bold;");
264 dettaglioRisultato.setText(
265 "Complimenti! Hai risposto correttamente.\n"
266 + "La parola era: \"" + result.getParolaCorretta() + "\"");
267 break;
268 case SCONFITTA:
269 titoloRisultato.setStyle(
270 "-fx-text-fill: #f85149; -fx-font-size: 26px; -fx-font-weight: bold;");
271 String winner = result.getNomeUtenteVincitore();
272 dettaglioRisultato.setText(
273 (winner != null ? winner + " ha risposto prima di te.\n" : "")
274 + "La parola corretta era: \"" + result.getParolaCorretta() + "\"");
275 break;
276 case PAREGGIO:
277 titoloRisultato.setStyle(
278 "-fx-text-fill: #e3b341; -fx-font-size: 26px; -fx-font-weight: bold;");
279 dettaglioRisultato.setText(
280 "Nessuno ha indovinato entro il tempo limite.\n"
281 + "La parola era: \"" + result.getParolaCorretta() + "\"");
282 break;
283 }
284
285 mostraPannelloRisultato();
286 }

298 private void gestisciErroreServer(String errorMsg) {
299 if (errorMsg != null && errorMsg.contains("disconnesso")) {
300 if (timelineTimer != null) timelineTimer.stop();
301 etichettaStato.setText("Partita annullata");
302 mostraFeedback("L'avversario si e' disconnesso. In attesa di nuovo avversario...");
303 Timeline delay = new Timeline(new KeyFrame(Duration.seconds(3),
304 new EventHandler<ActionEvent>() {
305 @Override
306 public void handle(ActionEvent event) {
307 dettaglioAttesa.setText("In attesa di un nuovo avversario...");
308 mostraAttesa();
309 connessione.invia(new Messaggio(Messaggio.Tipo.REQUEUE_REQUEST, null));
310 }
311 }));
312 delay.setCycleCount(1);
313 delay.play();
314 } else {
315 mostraFeedback(errorMsg != null ? errorMsg : "Errore sconosciuto.");
316 }
317 }
