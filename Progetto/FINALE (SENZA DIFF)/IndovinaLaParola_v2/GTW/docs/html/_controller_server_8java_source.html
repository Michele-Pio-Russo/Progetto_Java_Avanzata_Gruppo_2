386 private void populateTfTable(final Map<String, Double> tf) {
387 final ObservableList<TFEntry> data = FXCollections.observableArrayList();
388
389 // Stream per sorting e limit
390 tf.entrySet().stream()
391 .sorted((a, b) -> Double.compare(b.getValue(), a.getValue()))
392 .limit(500)
393 .forEach(e -> data.add(new TFEntry(e.getKey(), e.getValue())));
394
395 Platform.runLater(new Runnable() {
396 @Override
397 public void run() {
398 tfTableView.setItems(data);
399 }
400 });
