58 public GestoreClient(Socket socket, CoordinatoreGioco coordinator) {
59 this.socket = socket;
60 this.coordinator = coordinator;
61 try {
62 // ORDINE OBBLIGATORIO: ObjectOutputStream PRIMA di ObjectInputStream
63 // Senza il flush(), l'header OOS non viene inviato e il costruttore
64 // OIS sull'altro capo si blocca in attesa → deadlock
65 out = new ObjectOutputStream(socket.getOutputStream());
66 out.flush();
67 in = new ObjectInputStream(socket.getInputStream());
68 } catch (IOException e) {
69 LOGGER.log(Level.SEVERE, "Errore inizializzazione stream per: "
70 + socket.getRemoteSocketAddress(), e);
71 }
72 }

90 private void cicloAscolto() {
91 try {
92 while (!socket.isClosed()) {
93 Messaggio msg = (Messaggio) in.readObject();
94 smistaMessaggio(msg);
95 }
96 } catch (EOFException | java.net.SocketException e) {
97 // Disconnessione normale (client chiuso o rete interrotta)
98 LOGGER.info("Client disconnesso: "
99 + (nomeUtente != null ? nomeUtente : socket.getRemoteSocketAddress()));
100 } catch (Exception e) {
101 LOGGER.log(Level.WARNING, "Errore ricezione messaggio da: " + nomeUtente, e);
102 } finally {
103 coordinator.onClientDisconnesso(this);
104 close();
105 }
106 }

114 private void smistaMessaggio(Messaggio msg) {
115 switch (msg.getTipo()) {
116 case LOGIN_REQUEST:
117 coordinator.gestisciLogin(this, (PayloadAutenticazione) msg.getCarico());
118 break;
119 case REGISTER_REQUEST:
120 coordinator.gestisciRegistrazione(this, (PayloadAutenticazione) msg.getCarico());
121 break;
122 case CHALLENGE_ANSWER:
123 coordinator.gestisciRisposta(this, (String) msg.getCarico());
124 break;
125 case HISTORY_REQUEST:
126 coordinator.gestisciRichiestaStorico(this);
127 break;
128 case REQUEUE_REQUEST:
129 coordinator.gestisciNuovaRichiesta(this);
130 break;
131 default:
132 LOGGER.warning("Tipo messaggio non gestito: " + msg.getTipo());
133 break;
134 }
135 }
