93 private void configuraTabella() {
94 dateCol.setCellValueFactory(c ->
95 new SimpleStringProperty(c.getValue().getDataPartita()));
96 opponentCol.setCellValueFactory(c ->
97 new SimpleStringProperty(c.getValue().getAvversario()));
98 wordCol.setCellValueFactory(c ->
99 new SimpleStringProperty(c.getValue().getParolaCorretta()));
100 outcomeCol.setCellValueFactory(c ->
101 new SimpleStringProperty(formatOutcome(c.getValue().getEsito())));
102 timeCol.setCellValueFactory(c ->
103 new SimpleLongProperty(c.getValue().getTempoRispostaMs()).asObject());
104
105 outcomeCol.setCellFactory(col -> new TableCell<VoceStorico, String>() {
106 @Override
107 protected void updateItem(String item, boolean empty) {
108 super.updateItem(item, empty);
109 if (empty || item == null) {
110 setText(null);
111 setStyle("");
112 } else {
113 setText(item);
114 if (item.startsWith("Vinto")) {
115 setStyle("-fx-text-fill: #3fb950; -fx-font-weight: bold;");
116 } else if (item.startsWith("Perso")) {
117 setStyle("-fx-text-fill: #f85149; -fx-font-weight: bold;");
118 } else {
119 setStyle("-fx-text-fill: #e3b341; -fx-font-weight: bold;");
120 }
121 }
122 }
123 });
124
125 timeCol.setCellFactory(col -> new TableCell<VoceStorico, Long>() {
126 @Override
127 protected void updateItem(Long item, boolean empty) {
128 super.updateItem(item, empty);
129 if (empty || item == null) {
130 setText(null);
131 } else {
132 setText(item < 0 ? "\u2014" : String.valueOf(item));
133 }
134 }
135 });

146 @SuppressWarnings("unchecked")
147 private void gestisciMessaggio(final Messaggio msg) {
148 if (msg.getTipo() == Messaggio.Tipo.HISTORY_RESPONSE) {
149 final List<VoceStorico> entries = (List<VoceStorico>) msg.getCarico();
150 Platform.runLater(new Runnable() {
151 @Override
152 public void run() {
153 ObservableList<VoceStorico> data =
154 FXCollections.observableArrayList(entries);
155 historyTable.setItems(data);
156 }
157 });
158 }
159 }

168 @FXML

177 stageProprietario.setScene(new Scene(root, 820, 620));
178 stageProprietario.setTitle("IndovinaLaParola - " + nomeUtente);
179
180 connessione.invia(new Messaggio(Messaggio.Tipo.REQUEUE_REQUEST, null));
181 } catch (IOException e) {
182 LOGGER.log(Level.SEVERE, "Errore ritorno alla schermata di gioco", e);
183 }

192 private String formatOutcome(String esito) {
193 if (esito == null) return "\u2014";
194 if ("win".equalsIgnoreCase(esito)) return "Vinto";
195 if ("loss".equalsIgnoreCase(esito)) return "Perso";
196 if ("draw".equalsIgnoreCase(esito)) return "Pareggio";
197 return esito;
198 }
