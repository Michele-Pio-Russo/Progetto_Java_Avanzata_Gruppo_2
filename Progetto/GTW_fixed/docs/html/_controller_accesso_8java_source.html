148 private void gestisciMessaggioServer(final Messaggio msg) {
149 if (msg.getTipo() == Messaggio.Tipo.WAITING) {
150 final java.util.Queue<Messaggio> pending = new java.util.concurrent.ConcurrentLinkedQueue<>();
151 connessione.setMessageCallback(new Consumer<Messaggio>() {
152 @Override
153 public void accept(Messaggio m) {
154 pending.add(m);
155 }
156 });
157 Platform.runLater(new Runnable() {
158 @Override
159 public void run() {
160 pulsantePrincipale.setDisable(false);
161 vaiAGioco(
162 campoNomeUtente.getText().trim(),
163 connessione,
164 (String) msg.getCarico(),
165 pending);
166 }
167 });
168 return;
169 }
170
171 Platform.runLater(new Runnable() {
172 @Override
173 public void run() {
174 pulsantePrincipale.setDisable(false);
175 switch (msg.getTipo()) {
176 case LOGIN_RESPONSE:
177 gestisciRispostaLogin((RispostaAutenticazione) msg.getCarico());
178 break;
179 case REGISTER_RESPONSE:
180 gestisciRispostaRegistrazione((RispostaAutenticazione) msg.getCarico());
181 break;
182 case ERROR:
183 mostraErrore((String) msg.getCarico());
184 break;
185 default:
186 break;
187 }
188 }
189 });
190 }

240 @FXML
241 private void handlePrimary() {
242 clearError();
243 final String nomeUtente = campoNomeUtente.getText().trim();
244 final String password = passwordField.getText().trim();
245
246 if (nomeUtente.isEmpty() || password.isEmpty()) {
247 mostraErrore("Username e password sono obbligatori e non possono contenere solo spazi.");
248 return;
249 }
250
251 if (isRegistrationMode) {
252 if (nomeUtente.length() < 3) {
253 mostraErrore("Username deve avere almeno 3 caratteri.");
254 return;
255 }
256 if (password.length() < 4) {
257 mostraErrore("Password deve avere almeno 4 caratteri.");
258 return;
259 }
260 final String confirm = confirmField.getText().trim();
261 if (!password.equals(confirm)) {
262 mostraErrore("Le password non coincidono.");
263 return;
264 }
265 }
266
267 if (!connessione.isConnesso()) {
269 mostraErrore("Tentativo di connessione al server in corso...");
270 pulsantePrincipale.setDisable(true);
271 Thread t = new Thread(new Runnable() {
272 @Override
273 public void run() {
274 try {
275 connessione.connetti(serverIp, serverPort);
276 Platform.runLater(new Runnable() {
277 @Override
278 public void run() {
280 connectionLabel.setText("Connesso a " + serverIp + ":" + serverPort);
281 clearError();
282 inviaRichiesta(nomeUtente, password);
283 }
284 });
285 } catch (IOException e) {
286 Platform.runLater(new Runnable() {
287 @Override
288 public void run() {
290 pulsantePrincipale.setDisable(false);
291 mostraErrore("Server non raggiungibile. Accendi il server e riprova.");
292 }
293 });
294 }
295 }
296 });
297 t.setDaemon(true);
298 t.start();
299 return;
300 }

302 inviaRichiesta(nomeUtente, password);
303 pulsantePrincipale.setDisable(true);

359 private void vaiAGioco(final String nomeUtente, final ConnessioneServer conn, final String waitingMsg, final java.util.Queue<Messaggio> pendingMessages) {
360 try {
361 FXMLLoader loader = new FXMLLoader(
362 getClass().getResource("/fxml/game.fxml"));
363 Parent root = loader.load();
364 ControllerGioco gc = loader.getController();
365 gc.init(nomeUtente, conn, waitingMsg);
366
367 Messaggio m;
368 while ((m = pendingMessages.poll()) != null) {
369 gc.consegnaMessaggio(m);
370 }
371
372 Stage stage = (Stage) pulsantePrincipale.getScene().getWindow();
373 stage.setScene(new Scene(root, 820, 620));
374 stage.setTitle("IndovinaLaParola - " + nomeUtente);
375 stage.setResizable(true);
376 stage.setMinWidth(700);
377 stage.setMinHeight(520);
378 } catch (IOException e) {
379 LOGGER.log(Level.SEVERE, "Errore caricamento game.fxml", e);
380 mostraErrore("Errore interno: impossibile aprire la finestra di gioco.");
381 }
382 }
