169 public synchronized void gestisciRegistrazione(GestoreClient client, PayloadAutenticazione carico) {
170 String nome = carico.getNomeUtente() == null ? "" : carico.getNomeUtente().trim();
171 String pass = carico.getPassword() == null ? "" : carico.getPassword().trim();
172
173 if (nome.isEmpty() || pass.isEmpty()) {
174 client.invia(new Messaggio(Messaggio.Tipo.REGISTER_RESPONSE,
175 new RispostaAutenticazione(false, "Username e password non possono essere vuoti o contenere solo spazi.", null)));
176 return;
177 }
178 if (nome.length() < 3) {
179 client.invia(new Messaggio(Messaggio.Tipo.REGISTER_RESPONSE,
180 new RispostaAutenticazione(false, "Username deve avere almeno 3 caratteri.", null)));
181 return;
182 }
183 if (pass.length() < 4) {
184 client.invia(new Messaggio(Messaggio.Tipo.REGISTER_RESPONSE,
185 new RispostaAutenticazione(false, "Password deve avere almeno 4 caratteri.", null)));
186 return;
187 }
188 boolean ok = db.registraUtente(nome, pass);
189 if (ok) {
190 client.invia(new Messaggio(Messaggio.Tipo.REGISTER_RESPONSE,
191 new RispostaAutenticazione(true, "Registrazione completata. Ora accedi.", null)));
192 log("Registrato: " + carico.getNomeUtente());
193 } else {
194 client.invia(new Messaggio(Messaggio.Tipo.REGISTER_RESPONSE,
195 new RispostaAutenticazione(false, "Username gia' esistente.", null)));
196 }
197 }

234 private void aggiungiInListaAttesa(GestoreClient client) {
235 // Controlla che lo stesso username non sia già in lista d'attesa
236 // (evita che un giocatore giochi contro se stesso con due client)
237 for (GestoreClient g : giocatoriInAttesa) {
238 if (g.getNomeUtente() != null &&
239 g.getNomeUtente().equalsIgnoreCase(client.getNomeUtente())) {
240 client.invia(new Messaggio(Messaggio.Tipo.ERROR,
241 "Sei gia' in lista d'attesa con un altro client. Impossibile giocare contro te stesso."));
242 log("Accesso duplicato bloccato: " + client.getNomeUtente());
243 return;
244 }
245 }
246
247 client.invia(new Messaggio(Messaggio.Tipo.WAITING,
248 "In attesa dell'avversario... (" + (giocatoriInAttesa.size() + 1) + "/2)"));
249 giocatoriInAttesa.add(client);
250 log("In attesa: " + client.getNomeUtente()
251 + " (" + giocatoriInAttesa.size() + "/2)");
252
253 if (giocatoriInAttesa.size() >= 2) {
254 giocatore1 = giocatoriInAttesa.remove(0);
255 giocatore2 = giocatoriInAttesa.remove(0);
256 avviaSfida(giocatore1, giocatore2);
257 }

269 private void avviaSfida(GestoreClient p1, GestoreClient p2) {
270 if (analizzatore == null || !analizzatore.hasResults()) {
271 String err = "Il server non ha documenti analizzati. Impossibile avviare la sfida.";
272 log("ERRORE: " + err);
273 p1.invia(new Messaggio(Messaggio.Tipo.ERROR, err));
274 p2.invia(new Messaggio(Messaggio.Tipo.ERROR, err));
275 return;
276 }
277
278 // Estrae estratto e parola
279 String excerpt = analizzatore.extractRandomExcerpt(excerptWords);
280 String parola = analizzatore.selectWordFromExcerpt(excerpt, 2);
281 if (parola == null) {
282 log("Impossibile selezionare parola dall'estratto. Riprovo...");
283 excerpt = analizzatore.extractRandomExcerpt(excerptWords);
284 parola = analizzatore.selectWordFromExcerpt(excerpt, 1);
285 }
286 if (parola == null) {
287 log("ERRORE: nessuna parola idonea trovata.");
288 return;
289 }
290
291 // Cifra la parola con spostamento casuale 1–25
292 int spostamento = new Random().nextInt(25) + 1;
293 String encrypted = CifrarioCesare.cifra(parola, spostamento);
294
295 // Sostituisce la parola nell'estratto con quella cifrata tra parentesi quadre
296 // Pattern.quote() protegge da caratteri speciali regex nella parola
297 String textWithCipher = excerpt.replaceAll(
298 "(?i)\\b" + java.util.regex.Pattern.quote(parola) + "\\b", "[" + encrypted + "]");
299
300 // Salva lo stato della sfida in corso
301 parolaOriginaleCorrente = parola;
302 parolaCifrataCorrente = encrypted;
303 spostamentoCorrente = spostamento;
304 estrattoCorrente = textWithCipher;
305 tempoInizioSfida = System.currentTimeMillis();
306 sfidaAttiva = true;
307
308 List<String> encryptedList = Collections.singletonList(encrypted);
309 PayloadSfida carico = new PayloadSfida(0, textWithCipher, encryptedList, timeoutSeconds);
310
311 p1.invia(new Messaggio(Messaggio.Tipo.CHALLENGE_START, carico));
312 p2.invia(new Messaggio(Messaggio.Tipo.CHALLENGE_START, carico));
313
314 log("Sfida avviata: " + p1.getNomeUtente() + " vs " + p2.getNomeUtente()
315 + " | parola: " + parola + " | spostamento: " + spostamento);
316
317 // Timeout automatico con ScheduledExecutorService
318 // La lambda cattura 'this' per chiamare terminaSfida in modo synchronized
319 taskTimeout = scheduler.schedule(new Runnable() {
320 @Override
321 public void run() {
322 synchronized (CoordinatoreGioco.this) {
323 if (sfidaAttiva) {
324 log("Timeout scaduto per la sfida.");
325 terminaSfida(null, -1, true);
326 }
327 }
328 }
329 }, timeoutSeconds, TimeUnit.SECONDS);
330 }

343 public synchronized void gestisciRisposta(GestoreClient client, String guess) {
344 if (!sfidaAttiva) return;
345 if (client != giocatore1 && client != giocatore2) return;
346 if (guess == null) return;
347
348 long responseTime = System.currentTimeMillis() - tempoInizioSfida;
349
350 if (guess.trim().equalsIgnoreCase(parolaOriginaleCorrente)) {
351 // Risposta corretta → il client che ha risposto vince
352 terminaSfida(client, responseTime, false);
353 } else {
354 // Risposta errata → notifica solo questo client, può riprovare
355 client.invia(new Messaggio(Messaggio.Tipo.ERROR,
356 "Risposta errata. Riprova!"));
357 log(client.getNomeUtente() + " ha risposto erroneamente: " + guess);
358 }
359 }

369 private void terminaSfida(GestoreClient winner, long tempoRispostaMs, boolean isDraw) {
370 if (!sfidaAttiva) return;
371 sfidaAttiva = false;
372 if (taskTimeout != null) taskTimeout.cancel(false);
373
374 GestoreClient loser = (winner == giocatore1) ? giocatore2 : giocatore1;
375
376 if (!isDraw && winner != null) {
377 // Vittoria
378 winner.invia(new Messaggio(Messaggio.Tipo.CHALLENGE_RESULT,
379 new RisultatoSfida(winner.getNomeUtente(), parolaOriginaleCorrente, RisultatoSfida.Esito.VITTORIA)));
380 loser.invia(new Messaggio(Messaggio.Tipo.CHALLENGE_RESULT,
381 new RisultatoSfida(winner.getNomeUtente(), parolaOriginaleCorrente, RisultatoSfida.Esito.SCONFITTA)));
382
383 // Salva in transazione
384 log("DB: salvataggio vittoria in corso...");
385 if (db == null) {
386 log("DB: ERRORE - riferimento db e' null!");
387 } else {
388 db.saveChallengeWithResults(
389 estrattoCorrente, parolaCifrataCorrente, parolaOriginaleCorrente, spostamentoCorrente,
390 winner.getNomeUtente(), loser.getNomeUtente(), false, tempoRispostaMs);
391 }
392
393 log("Vince: " + winner.getNomeUtente() + " in " + tempoRispostaMs + "ms. Parola: " + parolaOriginaleCorrente);
394 } else {
395 // Pareggio / timeout
396 RisultatoSfida draw = new RisultatoSfida(null, parolaOriginaleCorrente, RisultatoSfida.Esito.PAREGGIO);
397 if (giocatore1 != null) giocatore1.invia(new Messaggio(Messaggio.Tipo.CHALLENGE_RESULT, draw));
398 if (giocatore2 != null) giocatore2.invia(new Messaggio(Messaggio.Tipo.CHALLENGE_RESULT, draw));
399
400 String u1 = giocatore1 != null ? giocatore1.getNomeUtente() : "";
401 String u2 = giocatore2 != null ? giocatore2.getNomeUtente() : "";
402 log("DB: salvataggio pareggio in corso... u1=" + u1 + " u2=" + u2);
403 if (db == null) {
404 log("DB: ERRORE - riferimento db e' null!");
405 } else {
406 db.saveChallengeWithResults(
407 estrattoCorrente, parolaCifrataCorrente, parolaOriginaleCorrente, spostamentoCorrente,
408 u1, u2, true, -1);
409 }
410
411 log("Pareggio/Timeout. Parola: " + parolaOriginaleCorrente);
412 }
413
414 giocatore1 = null;
415 giocatore2 = null;
416 }

425 public synchronized void onClientDisconnesso(GestoreClient client) {
