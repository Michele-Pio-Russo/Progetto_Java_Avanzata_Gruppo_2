161 private void insertUserIfAbsent(String nomeUtente, String password, String ruolo)
162 throws SQLException {
163 // try-with-resources per PreparedStatement
164 try (PreparedStatement check = connessione.prepareStatement(
165 "SELECT COUNT(*) FROM users WHERE nomeUtente = ?")) {
166 check.setString(1, nomeUtente);
167 try (ResultSet rs = check.executeQuery()) {
168 if (rs.next() && rs.getInt(1) == 0) {
169 try (PreparedStatement ins = connessione.prepareStatement(
170 "INSERT INTO users(nomeUtente, password_hash, ruolo) VALUES(?,?,?)")) {
171 ins.setString(1, nomeUtente);
172 ins.setString(2, hashPassword(password));
173 ins.setString(3, ruolo);
174 ins.executeUpdate();
175 LOGGER.info("Account predefinito creato: " + nomeUtente);
176 }
177 }
178 }

270 public void saveChallengeWithResults(
271 String estrattoTesto, String encryptedWord, String originalWord, int spostamento,
272 String winnerUser, String loserUser, boolean draw, long tempoRispostaMs) {
273
274 if (connessione == null) {
275 LOGGER.severe("saveChallengeWithResults: nessuna connessione al database "
276 + "(probabile driver SQLite non caricato). Risultato NON salvato.");
277 return;
278 }
279
280 try {
281 // Inizio transazione esplicita
282 connessione.setAutoCommit(false);
283
284 try {
285 // 1. Salva la sfida
286 int idSfida = insertChallenge(estrattoTesto, encryptedWord, originalWord, spostamento);
287
288 // 2. Salva i risultati
289 if (draw) {
290 int id1 = getUserId(winnerUser != null ? winnerUser : "");
291 int id2 = getUserId(loserUser != null ? loserUser : "");
292 if (id1 > 0) insertResult(idSfida, id1, "draw", -1);
293 if (id2 > 0) insertResult(idSfida, id2, "draw", -1);
294 } else {
295 int winnerId = getUserId(winnerUser);
296 int loserId = getUserId(loserUser);
297 if (winnerId > 0) insertResult(idSfida, winnerId, "win", tempoRispostaMs);
298 if (loserId > 0) insertResult(idSfida, loserId, "loss", -1);
299 }
300
301 // Commit — tutto ok
302 connessione.commit();
303 LOGGER.info("Transazione sfida+risultati salvata. ChallengeId=" + idSfida);
304
305 } catch (SQLException e) {
306 // Rollback in caso di errore
307 try { connessione.rollback(); } catch (SQLException ex) { /* ignore */ }
308 LOGGER.log(Level.SEVERE, "Rollback transazione sfida: " + e.getMessage(), e);
309 } finally {
310 connessione.setAutoCommit(true);
311 }
312
313 } catch (SQLException e) {
314 LOGGER.log(Level.SEVERE, "Errore gestione transazione", e);
315 } catch (RuntimeException e) {
316 // Difesa extra: qualunque errore inatteso (es. NPE) non deve
317 // propagarsi al thread di rete del GestoreClient, altrimenti
318 // il client che ha inviato CHALLENGE_ANSWER verrebbe disconnesso.
319 LOGGER.log(Level.SEVERE, "Errore inatteso nel salvataggio risultati sfida", e);
320 }
321 }

381 public List<VoceStorico> getHistory(String nomeUtente) {
382 List<VoceStorico> list = new ArrayList<>();
383 String sql =
384 "SELECT c.played_at, c.original_word, r.esito, r.response_time_ms," +
385 " (SELECT u2.nomeUtente FROM results r2" +
386 " JOIN users u2 ON r2.user_id = u2.user_id" +
387 " WHERE r2.challenge_id = c.challenge_id AND u2.nomeUtente != ?) AS avversario " +
388 "FROM results r " +
389 "JOIN challenges c ON r.challenge_id = c.challenge_id " +
390 "JOIN users u ON r.user_id = u.user_id " +
391 "WHERE u.nomeUtente = ? " +
392 "ORDER BY c.played_at DESC";
393 try (PreparedStatement ps = connessione.prepareStatement(sql)) {
394 ps.setString(1, nomeUtente);
395 ps.setString(2, nomeUtente);
396 try (ResultSet rs = ps.executeQuery()) {
397 while (rs.next()) {
398 String avversario = rs.getString("avversario");
399 list.add(new VoceStorico(
400 rs.getString("played_at"),
401 avversario != null ? avversario : "?",
402 rs.getString("esito"),
403 rs.getLong("response_time_ms"),
404 rs.getString("original_word")
405 ));
406 }
407 }
408 } catch (SQLException e) {
409 LOGGER.log(Level.SEVERE, "Errore getHistory per: " + nomeUtente, e);

450 public List<UserStats> getAllStats() {
451 List<UserStats> stats = new ArrayList<>();
452 String sql =
453 "SELECT u.nomeUtente," +
454 " SUM(CASE WHEN r.esito = 'win' THEN 1 ELSE 0 END) AS wins," +
455 " COUNT(*) AS games_played," +
456 " CAST(AVG(CASE WHEN r.response_time_ms >= 0 THEN r.response_time_ms END) AS INTEGER) AS avg_ms " +
457 "FROM results r " +
458 "JOIN users u ON r.user_id = u.user_id " +
459 "GROUP BY r.user_id " +
460 "ORDER BY wins DESC";
461 try (Statement stmt = connessione.createStatement();
462 ResultSet rs = stmt.executeQuery(sql)) {
463 while (rs.next()) {
464 stats.add(new UserStats(
465 rs.getString("nomeUtente"),
466 rs.getInt("wins"),
467 rs.getInt("games_played"),
468 rs.getLong("avg_ms")
469 ));
470 }
471 } catch (SQLException e) {
472 LOGGER.log(Level.SEVERE, "Errore getAllStats", e);
473 }

486 public static String hashPassword(String password) {
487 try {
488 MessageDigest md = MessageDigest.getInstance("SHA-256");
489 byte[] hash = md.digest(password.getBytes());
490 StringBuilder sb = new StringBuilder();
491 for (byte b : hash) {
492 sb.append(String.format("%02x", b));
493 }
494 return sb.toString();
495 } catch (NoSuchAlgorithmException e) {
496 throw new RuntimeException("SHA-256 non disponibile nella JVM", e);
497 }
498 }
